cmake_minimum_required(VERSION 3.21)
cmake_policy(SET CMP0091 NEW)
# Default review never enables a compiler. No DLL or executable project target.
project(P1CppEntryStep32 LANGUAGES NONE)
option(P1_ENABLE_OBJECT_CHECK "Future real-SDK object-only check; no DLL link" OFF)
set(P1_CRT "MT" CACHE STRING "Step32 sample route supports only MT")
set_property(CACHE P1_CRT PROPERTY STRINGS MT)
set(P1_SDK_ROOT "" CACHE PATH "Real local ProToolkit SDK root; never redistributed")
set(P1_EXPECTED_COMPILER_VERSION "19.44.35229.0" CACHE STRING "Reported file version for identity matching; not official support certification")
set(P1_EXPECTED_TOOLSET_VERSION "14.44.35207" CACHE STRING "Reported installed MSVC toolset directory")
set(P1_EXPECTED_WINDOWS_SDK_VERSION "10.0.26100.0" CACHE STRING "Reported Windows SDK version")
if(DEFINED P1_DOCUMENTED_MSVC_VERSION)
  message(FATAL_ERROR "Use P1_EXPECTED_COMPILER_VERSION for identity only; no official patch certification is implied")
endif()
if(NOT P1_CRT STREQUAL "MT")
  message(FATAL_ERROR "Step32 follows the reported MT sample; MD is not enabled in this template")
endif()
if(P1_ENABLE_OBJECT_CHECK)
  if(NOT CMAKE_HOST_WIN32)
    message(FATAL_ERROR "Object check requires a real Windows/MSVC host; no SDK substitutes")
  endif()
  if(NOT CMAKE_GENERATOR STREQUAL "Visual Studio 17 2022" OR NOT CMAKE_GENERATOR_PLATFORM STREQUAL "x64")
    message(FATAL_ERROR "Use the VS2022 generator with -A x64")
  endif()
  if(NOT CMAKE_GENERATOR_TOOLSET STREQUAL "v143,version=${P1_EXPECTED_TOOLSET_VERSION}")
    message(FATAL_ERROR "Select -T v143,version=<expected toolset directory>")
  endif()
  if(NOT CMAKE_SYSTEM_VERSION STREQUAL P1_EXPECTED_WINDOWS_SDK_VERSION)
    message(FATAL_ERROR "Select -DCMAKE_SYSTEM_VERSION=<expected Windows SDK version>")
  endif()
  foreach(header ProToolkit.h ProToolkitDll.h)
    if(NOT EXISTS "${P1_SDK_ROOT}/includes/${header}")
      message(FATAL_ERROR "Missing real SDK includes/${header}")
    endif()
  endforeach()
  if(NOT P1_EXPECTED_COMPILER_VERSION MATCHES "^[0-9]+(\\.[0-9]+)+$")
    message(FATAL_ERROR "Expected compiler version must be numeric")
  endif()
  set(CMAKE_CONFIGURATION_TYPES "Release" CACHE STRING "Object check uses Release only" FORCE)
  # Compiler identification is CMake-controlled; try_compile uses static archives.
  set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
  enable_language(CXX)
  if(NOT MSVC OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "MSVC" OR NOT MSVC_TOOLSET_VERSION EQUAL 143)
    message(FATAL_ERROR "Expected Microsoft MSVC v143")
  endif()
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "x64")
    message(FATAL_ERROR "Expected Windows x64 compiler")
  endif()
  if(NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL P1_EXPECTED_COMPILER_VERSION)
    message(FATAL_ERROR "Actual compiler differs from reported file-version candidate; investigate before proceeding")
  endif()
  if(NOT CMAKE_VS_PLATFORM_TOOLSET_VERSION VERSION_EQUAL P1_EXPECTED_TOOLSET_VERSION)
    message(FATAL_ERROR "Actual toolset differs from selected installation")
  endif()
  if(NOT CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION VERSION_EQUAL P1_EXPECTED_WINDOWS_SDK_VERSION)
    message(FATAL_ERROR "Actual Windows SDK differs from selected version")
  endif()
  add_library(p1_entry_object OBJECT src/entry.cpp)
  target_include_directories(p1_entry_object PRIVATE "${P1_SDK_ROOT}/includes")
  target_compile_definitions(p1_entry_object PRIVATE _WSTDIO_DEFINED PRO_MACHINE=36 PRO_OS=4)
  target_compile_options(p1_entry_object PRIVATE /GS /fp:precise)
  set_property(TARGET p1_entry_object PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
  # No PTC/system libraries are linked by an object-only target.
endif()
configure_file(cmake/build-plan.json.in build-plan.json @ONLY)
add_custom_target(review ALL
  COMMAND "${CMAKE_COMMAND}" -E echo "REVIEW_ONLY: v143, MT, sample macros; no DLL link or deployment"
  VERBATIM)
