cmake_minimum_required(VERSION 3.21)
cmake_policy(SET CMP0091 NEW)
# Review is compiler-free. There is intentionally no DLL or executable target.
project(P1CppEntryReview LANGUAGES NONE)
set(P1_CRT "MT" CACHE STRING "Provisional CRT profile: MT or MD")
set_property(CACHE P1_CRT PROPERTY STRINGS MT MD)
option(P1_ENABLE_OBJECT_CHECK "Future real-header object-only check; no linking" OFF)
set(P1_SDK_ROOT "" CACHE PATH "Real local ProToolkit SDK root (never distributed)")
set(P1_DOCUMENTED_MSVC_VERSION "" CACHE STRING "Exact MSVC compiler version selected from applicable PTC support evidence")
if(P1_CRT STREQUAL "MT")
  set(P1_CANDIDATE_LIBRARY "protk_dll_NU.lib")
  set(P1_RUNTIME "MultiThreaded")
elseif(P1_CRT STREQUAL "MD")
  set(P1_CANDIDATE_LIBRARY "protk_dllmd_NU.lib")
  set(P1_RUNTIME "MultiThreadedDLL")
else()
  message(FATAL_ERROR "P1_CRT must be MT or MD; do not mix CRT/library families")
endif()
if(P1_ENABLE_OBJECT_CHECK)
  if(NOT CMAKE_HOST_WIN32)
    message(FATAL_ERROR "Object check requires a real Windows/MSVC host; no SDK substitutes")
  endif()
  if(NOT P1_DOCUMENTED_MSVC_VERSION MATCHES "^[0-9]+(\\.[0-9]+)+$")
    message(FATAL_ERROR "Missing exact documented MSVC compiler version; do not guess the toolset")
  endif()
  foreach(header ProToolkit.h ProToolkitDll.h)
    if(NOT EXISTS "${P1_SDK_ROOT}/includes/${header}")
      message(FATAL_ERROR "Missing real SDK includes/${header}")
    endif()
  endforeach()
  # Restrict try_compile to static archives; compiler identification is CMake-controlled.
  set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
  enable_language(CXX)
  if(NOT MSVC OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
    message(FATAL_ERROR "Only the documented Microsoft MSVC compiler is eligible")
  endif()
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "x64")
    message(FATAL_ERROR "This candidate requires Windows x64")
  endif()
  if(NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL P1_DOCUMENTED_MSVC_VERSION)
    message(FATAL_ERROR "Detected compiler differs from documented selection")
  endif()
  if(CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_CONFIGURATION_TYPES "Release" CACHE STRING "Only Release candidate" FORCE)
  elseif(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "Object check requires CMAKE_BUILD_TYPE=Release")
  endif()
  add_library(p1_entry_object OBJECT src/entry.cpp)
  target_include_directories(p1_entry_object PRIVATE "${P1_SDK_ROOT}/includes")
  set_property(TARGET p1_entry_object PROPERTY MSVC_RUNTIME_LIBRARY "${P1_RUNTIME}")
  target_compile_options(p1_entry_object PRIVATE /W3 /EHsc)
  # No target_link_libraries, DLL target, post-build step, or deployment.
endif()
configure_file(cmake/build-plan.json.in build-plan.json @ONLY)
add_custom_target(review ALL
  COMMAND "${CMAKE_COMMAND}" -E echo "REVIEW_ONLY: ${P1_CRT}, candidate ${P1_CANDIDATE_LIBRARY}; no link or deployment"
  VERBATIM)
